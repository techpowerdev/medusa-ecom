# ข้อมูลสังเคราะห์ (ANALYZE แล้ว): group_buy_campaign 200, group_buy_group 20k, group_buy_membership 200k, evoucher_code 200k (100 pool),

## SQL ที่ใช้
\echo '-- Q1 store GET /store/group-buy/groups/:token'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT * FROM group_buy_group WHERE join_token = 'tok500' AND deleted_at IS NULL;
\echo '-- Q2 store GET /store/group-buy/campaigns/active (pool groups)'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT * FROM group_buy_group WHERE campaign_id IN ('gbcamp_1','gbcamp_2') AND kind = 'pool' AND status = 'open' AND ends_at > now() AND deleted_at IS NULL;
\echo '-- Q3 store GET /store/customers/me/group-buy/memberships'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT * FROM group_buy_membership WHERE customer_id = 'cus_42' AND deleted_at IS NULL ORDER BY joined_at DESC LIMIT 20;
\echo '-- Q4 store GET /store/customers/me/evouchers'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT * FROM evoucher_code WHERE customer_id = 'cus_42' AND status IN ('allocated','delivered') AND deleted_at IS NULL;
\echo '-- Q5 store GET /store/flash-sales/active + checkout hook (config by campaign)'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT * FROM flash_sale_config WHERE campaign_id IN ('procamp_7') AND deleted_at IS NULL;
\echo '-- Q6 checkout flash usage upsert target'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) UPDATE flash_sale_usage SET units = units + 1 WHERE campaign_id = 'procamp_2' AND customer_id = 'cus_1' AND deleted_at IS NULL;
\echo '-- Q7 checkout flash reservation by cart'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) UPDATE flash_sale_reservation SET released_at = now() WHERE cart_id = 'cart_1' AND released_at IS NULL AND deleted_at IS NULL;
\echo '-- Q8 evoucher claim (order placed)'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT id FROM evoucher_code WHERE pool_id = 'evpool_3' AND status = 'available' AND deleted_at IS NULL LIMIT 5 FOR UPDATE SKIP LOCKED;
\echo '-- Q9 affiliate attribution by order'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT * FROM affiliate_attribution WHERE order_id = 'order_af_500' AND deleted_at IS NULL;
\echo '-- Q10 job group-buy-deadline (every minute): groups past ends_at'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT id FROM group_buy_group WHERE status = 'open' AND ends_at <= now() AND deleted_at IS NULL ORDER BY ends_at LIMIT 50;
\echo '-- Q11 job group-buy-deadline (every minute): memberships awaiting_payment past pay_deadline_at'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT m.id FROM group_buy_membership m JOIN group_buy_group g ON g.id = m.group_id AND g.deleted_at IS NULL WHERE m.status = 'awaiting_payment' AND m.deleted_at IS NULL AND g.pay_deadline_at <= now() ORDER BY m.joined_at LIMIT 50;
\echo '-- Q12 affiliate reverse-commission / maturity: ledger entries by order'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT * FROM affiliate_commission_entry WHERE order_id = 'order_af_500' AND deleted_at IS NULL;
\echo '-- Q13 affiliate maturity: reversed entries of matured orders'
EXPLAIN (ANALYZE, COSTS OFF, TIMING OFF, SUMMARY ON) SELECT * FROM affiliate_commission_entry WHERE status = 'reversed' AND order_id IN ('order_af_10','order_af_20') AND deleted_at IS NULL;
